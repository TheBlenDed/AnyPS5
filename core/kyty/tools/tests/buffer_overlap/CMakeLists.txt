cmake_minimum_required(VERSION 3.25)
project(KytyBufferOverlapTest LANGUAGES CXX)
include(CTest)

set(renderer "${CMAKE_CURRENT_SOURCE_DIR}/../../../graphics/host_gpu/renderer")
set(BUFFER_CACHE_SOURCE "${renderer}/cache/bufferCache.cpp" CACHE FILEPATH "Buffer cache implementation under test")
option(EXPECT_DIRECTED_RESERVE "Expect directed growth at a multi-buffer boundary" ON)
file(READ "${BUFFER_CACHE_SOURCE}" source)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${BUFFER_CACHE_SOURCE}")
string(FIND "${source}" "BufferCache::OverlapResult BufferCache::ResolveOverlaps" begin)
string(FIND "${source}" "void BufferCache::JoinOverlap" end)
if(begin LESS 0 OR end LESS_EQUAL begin)
    message(FATAL_ERROR "Cannot locate ResolveOverlaps in ${BUFFER_CACHE_SOURCE}")
endif()
math(EXPR length "${end} - ${begin}")
string(SUBSTRING "${source}" ${begin} ${length} implementation)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/RangeImplementation.inc" "${implementation}")

add_executable(kyty_buffer_overlap BufferOverlap.cpp)
target_compile_features(kyty_buffer_overlap PRIVATE cxx_std_20)
target_compile_definitions(kyty_buffer_overlap PRIVATE CANDIDATE=$<BOOL:${EXPECT_DIRECTED_RESERVE}>)
target_include_directories(kyty_buffer_overlap PRIVATE "${CMAKE_CURRENT_BINARY_DIR}"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../3rdparty/Vulkan-Headers/include")
if(WIN32)
    set(defaultLoader vulkan-1)
else()
    set(defaultLoader vulkan)
endif()
set(VULKAN_LOADER "${defaultLoader}" CACHE STRING "Vulkan loader library or import library")
target_link_libraries(kyty_buffer_overlap PRIVATE "${VULKAN_LOADER}")
foreach(mode IN ITEMS right left third lower-bound upper-bound shapes)
    add_test(NAME range_${mode} COMMAND kyty_buffer_overlap ${mode})
endforeach()
